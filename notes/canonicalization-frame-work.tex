Following \cite{dym2024equivariant}, we make the following definitions in the context of canonicalization frames.

\begin{definition}
    Given a $G$-space $X$, a \emph{weighted canonicalization frame} is a function $\nu_\bullet : X \to \PP(X)$ where $\PP(X)$ is the space of Borel probability measures on $X$.
\end{definition}

\begin{definition}\label{def:weighted-canon-frame}
    A weighted canonicalization frame $\nu_\bullet$ is \emph{orbital} if $\nu_x(X \sminus Gx) = 0$ for all $x \in X$.
\end{definition}


\begin{remark}
    As per \cite{dym2024equivariant}, definition \ref{def:weighted-canon-frame} makes sense for arbitrary measures, but we will be primarily concerned with finite atomic measures of the form $\nu_x = \sum_{i} \alpha_i \delta_{x_i}$ where $\alpha_i \in [0,1]$ and $\sum_i \alpha_i = 1$.
\end{remark}


\begin{definition}
    A weighted canonicalization frame $\nu_\bullet$ is \emph{invariant} if $\nu_gx = \nu_x$ for all $x \in X$ and $g \in G$.
\end{definition}

% Going forward, we will assume that all weighted canonicalizations are orbital and invariant and thus not specify them as such.

\begin{proposition}
    Let $\nu_\bullet$ be an invariant weighted canonicalization frame.
    Then for any Borel function $f : X \to Y$, the function $\II_\nu (f)$ given by
    \[
        \II_\nu (f) (x) := \int_X f d \nu_x
    \]
    is $G$-invariant.
    \begin{proof}
        Let $g \in G$ and $x \in X$, and we simply compute
        \[
            \II_\nu (f) (gx)
            = \int_X f d \nu_{gx}
            = \int_X f d \nu_x
            = \II_\nu (f) (x).
        \]
    \end{proof}
\end{proposition}

\begin{definition}
    An invariant orbital weighted canonicalization frame $\nu_\bullet$ is continuous at $x$ if for any sequence $\seq{x} \sub X$ converging to $x$, we have $\nu_{x_n} \wto \nu_x$, i.e. for all $f : X \to \R$
    \[
        \int_X f d \nu_{x_n} \to \int_X f d \nu_x.
    \]
    A \emph{robust canonicalization frame} is a continuous invariant orbital weighted canonicalization frame.
\end{definition}

\begin{theorem}\label{thm:canon-frame-cont-iff-preserves-cont}
    A weighted canonicalization frame $\nu_\bullet$ is continuous if and only if $\II_\nu$ preserves continuity.
    \begin{proof}
        First assume that $\nu_\bullet$ is continuous and let $x \in X$ with $\seq{x} \sub X$ such that $x_n \to x$.
        Then for any map $f : X \to \R$, we immediately have
        \[
            \II_\nu (f) (x_n)
            = \int_X f d \nu_{x_n}
            \to \int_X f d \nu_x
            = \II_\nu (f) (x)
        \]
        so we conclude that $\II_\nu$ preserves continuity.
        
        Now instead assume that $\II_\nu$ preserves continuity, let $f : X \to \R$ be a Borel function, and let $x \in X$ with $\seq{x} \sub X$ such that $x_n \to x$.
        Then by continuity of $\II_\nu (f)$ we have
        \[
            \int_X f d \nu_{x_n}
            = \II_\nu (f) (x_n)
            \to \II_\nu (f) (x)
            = \int_X f d \nu_x
        \]
        so we are done.
    \end{proof}
\end{theorem}


Similar to theorem \ref{thm:frame-equivalence}, we have a correspondence between weighted frames and weighted canonicalization frames.

\begin{theorem}\label{thm:weight-frame-equivalence}
    Let $X$ be a normal $G$-space.
    Then every weighted frame induces a weighted canonicalization frame and vice versa (though not necessarily via a bijection).
    In particular, this correspondence sends weakly equivariant weighted frames to invariant orbital weighted canonicalization frames and weighted canonicalization frames to equivariant weighted frames, and both directions preserve the continuity of the frame.
    \begin{proof}
        To begin, let $\mu_\bullet$ be a weakly equivariant weighted frame.
        We define a weighted canonicalization frame $\nu_\bullet$ for Borel sets $E \sub X$ by
        \[
            \nu_x (E)
            = \Bar{\mu}_x(\set{g \in G : \inv{g}x \in E}).
        \]
        Then $\nu_\bullet$ is orbital since
        \[
            \nu_x (X \sminus Gx)
            = \Bar{\mu}_x(\set{g \in G : \inv{g}x \not\in Gx})
            = \Bar{\mu}_x(\emp) = 0
        \]
        and invariant since
        \begin{align*}
            \nu_{hx} (E)
            &= \Bar{\mu}_{hx}(\set{g \in G : \inv{g}hx \in E}) \\
            &= \Bar{\mu}_x(\inv{h}\set{g \in G : \inv{g}hx \in E}) \\
            &= \Bar{\mu}_x(\set{g \in G : \pinv{hg}hx = \inv{g}x \in E}) \\
            &= \nu_x (E).
        \end{align*}
        Now assume $\mu_\bullet$ is continuous, and we wish to show that $\II_\nu (f)$ is continuous for any continuous $f$.
        To see this, first let $E \sub X$ be a Borel set and observe
        \[
            \int_X \chi_E d \nu_x
            = \nu_x (E)
            = \Bar{\mu}_x (\set{g \in G : \inv{g}x \in E})
            = \int_G \chi_E (\inv{g} x) d \Bar{\mu}_x (g).
        \]
        Since any Borel function $f : X \to \R$ can be approximated by simple functions, the Lebesgue Dominated Convergence Theorem gives us that 
        \[
            \int_X f d\nu_x = \int_G f(\inv{g} x) d \Bar{\mu}_x (g).
        \]
        But recall by lemma \ref{lem:weighted-frame-integral-equivalence} that 
        \[
            \int_G f(\inv{g} x) d \Bar{\mu}_x (g)
            = \int_G f(\inv{g} x) d \mu_x (g)
        \]
        so we have that $\II_\nu (f) = \II_\mu (f)$ and hence $\nu_\bullet$ is continuous by theorem \ref{thm:canon-frame-cont-iff-preserves-cont}.
        
        For the other direction, let $\nu_\bullet$ be an invariant orbital weighted canonicalization frame.
        We define a weighted frame $\mu_\bullet$ as follows.
        \footnote{The idea for this construction was suggested in part by Claude via \href{https://claude.ai/share/d975e47f-d488-4821-8653-ae7bbac49887}{this chat}.}
        To begin, fix some $x \in X$ and define a linear function $T_x : \cont{G}{\R} \to \cont{G/G_x}{\R}$ via
        \[
            T_x(f)(gG_x) := \int_{G_x} f(\pinv{gs}) ds
        \]
        where this integral is with respect to the unique left-invariant normalized Haar measure on $G_x$.
        To verify that this is a well-defined function, let $g' \in G$ such that $g'G_x = gG_x$, i.e. $g' = gs'$ for some $s' \in G_x$.
        Then we have
        \[
            T_x(f)(g'G_x)
            = \int_{G_x} f(\pinv{g's}) ds
            = \int_{G_x} f(\pinv{gs's}) ds
            = \int_{G_x} f(\pinv{gs}) ds
            = T_x(f)(gG_x)
        \]
        where the third equality is by the left-invariance of the Haar measure.
        Observe that $T_x$ is also linear by the linearity of integration.
        Using $T_x$, we now define a linear functional $L_x : \cont{G/G_x}{\R} \to \R$ via
        \[
            L_x(f) := \int_{Gx} f \circ q_x d \nu_x
        \]
        where $q_x : Gx \to G/G_x$ is the quotient map.
        Observe that $L_x$ is clearly linear, again by the linearity of integration.
        Then $L_x \circ T_x : \cont{G;\R} \to \R$ is a linear functional since $L_x$ and $T_x$ are both linear, and $G$ is compact, so the Riesz-Markov-Kakutani Representation Theorem gives us that there is a unique Borel measure $\mu_x$ on $G$ such that for any Borel function $f : G \to R$,
        \[
            \int_G f d \mu_x
            = (L_x \circ T_x) (f).
            %= \int_{Gx} \int_{G_x} f (\pinv{q_x(c)s}) ds d \nu_x (c).
        \]
        Define $\mu_\bullet$ as the collection of these $\mu_x$'s, and note that $\mu_x$ is indeed a probability measure since
        \[
            T_x(1)(gG_x)
            = \int_{G_x} 1 ds
            = 1
        \]
        so
        \[
            \mu_x (G)
            = \int_G 1 d\mu_x
            = (L_x \circ T_x) (1)
            = \int_{Gx} (T_x(1) \circ q_x) d \nu_x 
            = \int_{Gx} 1 d \nu_x 
            = 1
        \]
        To see that $\mu_\bullet$ is equivariant, let $h \in G$ and $x \in X$.
        Then we compute
        \begin{align*}
            \int_G f d \mu_{hx}
            &= \int_{G hx} T_{hx} (f) (q_{hx}(c)) d \nu_{hx} (c) \\
            &= \int_{G x} T_{hx} (f) (q_{hx}(c)) d \nu_{x} (c) \\
            &= \int_{G x} \int_{G_{hx}} f(\pinv{gs}) ds d \nu_x (c) \tag{$\exists g \in G, c = ghx \implies q_{hx}(c) = gG_{hx}$} \\
            &= \int_{G x} \int_{h G_x \inv{h}} f(\pinv{gs}) ds d \nu_x (c) \\
            &= \int_{G x} \int_{G_x} f(\pinv{ghs\inv{h}}) ds d \nu_x (c) \\
            &= \int_{G x} \int_{G_x} f(h\pinv{(gh)s}) ds d \nu_x (c) \\
            &= \int_{G x} (T_x(f \circ (h \cdot -)) (q_x(c))) d \nu_x (c) \tag{$c = ghx \implies q_x(c) = ghG_x$} \\
            &= \int_G (f \circ (h \cdot -)) d \mu_x \\
            &= \int_G f d h_* \mu_x
        \end{align*}
        and we thus conclude that $\mu_{hx} = h_* \mu_x$.
        Lastly, assume $\nu_\bullet$ is continuous, and we wish to show that $\II_\mu (f)$ is continuous for any map $f : X \to \R$.
        To see this, we simply compute
        \begin{align*}
            \II_\mu (f) (x)
            &= \int_G f (\inv{g} x) d \mu_x \\
            &= \int_G \inv{f}_x (g) d \mu_x \tag{$\inv{f}_x (g) := f(\inv{g} x)$} \\
            &= (L_x \circ T_x) (\inv{f}_x) \\
            &= \int_{Gx} (T_x (\inv{f}_x) \circ q_x) (c)  d \nu_x(c) \\
            &= \int_{Gx} \int_{G_x} \inv{f}_x (\pinv{g s}) ds d \nu_x(c) \tag{$c = gx$} \\
            &= \int_{Gx} \int_{G_x} f (g s x) ds d \nu_x (c) \\
            &= \int_{Gx} \int_{G_x} f (g x) ds d \nu_x (c) \\
            &= \int_{Gx} f (c) d \nu_x (c) \\
            &= \int_X f (c) d \nu_x (c) \\
            &= \II_\nu (f).
        \end{align*}
        This means that $\II_\mu$ preserves continuity, so we have that $\mu_\bullet$ is continuous by theorem \ref{thm:frame-cont-iff-preserves-cont}.

        Lastly, we will show that each direction of this process is (essentially) invertible.
        First, consider some weakly equivariant weighted frame $\mu_\bullet$; let $\nu_\bullet$ be the invariant orbital weight canonicalization frame induced by $\mu_\bullet$; and let $\lambda_\bullet$ be the equivariant weighted frame induced by $\nu_\bullet$.
        We now wish to show that $\lambda_x = \Bar{\mu}_x$ for any $x \in X$, so let $f : G \to \R$ be Borel and observe
        \begin{align*}
            \int_G f d \lambda_x
            &= \int_{Gx} (T_x (f) \circ q_x) d \nu_x \\
            &= \int_G (T_x (f) \circ q_x) (\inv{g} x) d \Bar{\mu}_x \\
            &= \int_G (T_x (f) \circ q_x) (\inv{g} x) d \mu_x \tag{lemma \ref{lem:weighted-frame-integral-equivalence}} \\
            &= \int_G (T_x (f) \circ q_x) (\inv{g} x) d \mu_x \\
            &= \int_G \int_{G_x} f (\pinv{\inv{g} s}) ds d \mu_x \\
            &= \int_G \int_{G_x} f (\inv{s} g) ds d \mu_x \\
            &= \int_G \int_{G_x} f (s g) ds d \mu_x \\
            &= \int_G f d \Bar{\mu}_x.
        \end{align*}
        Since $f$ was arbitrary, we have that $\lambda_x = \Bar{\mu}_x$.
        
        For the other direction, consider some invariant orbital weighted canonicalization frame $\nu_\bullet$; let $\mu_\bullet$ be the equivariant weight frame induced by $\nu_\bullet$; and let $\upsilon_\bullet$ be the invariant orbital weighted canonicalization frame induced by $\mu_\bullet$.
        We now wish to show that $\upsilon_x = \nu_x$ for any $x \in X$, so let $f : X \to \R$ be Borel and observe
        \begin{align*}
            \int_X f d \upsilon_x
            &= \int_G f (\inv{g} x) d \Bar{\mu}_x \\
            &= \int_G f (\inv{g} x) d \mu_x \\
            &= \II_{\mu} (f) (x) \\
            &= \II_{\nu} (f) (x) \\
            &= \int_X f d \nu_x.
        \end{align*}
        Since $f$ was arbitrary, we have that $\upsilon_x = \nu_x$.
    \end{proof}
\end{theorem}


\begin{remark}
    If the action of $G$ is free, the correspondence in theorem \ref{thm:frame-equivalence} is much simpler since every integral over $G_x$ reduces to a single value.
\end{remark}

\begin{theorem}[Richter-Tchakaloff]\label{thm:richter-tchakaloff}
    (Theorem 1.24 of \cite{schmudgen2017momentproblem})
    Let $(X,\mu)$ be a measure space and let $V \sub L^1(X)$ be a finite dimensional subspace with $\dim V = k$.
    Then there exists a $k$-atomic measure $\nu = \sum_{i \leq k} \alpha_i \delta_{x_i}$ on $X$ such that for every $f \in V$,
    \[
        \int_X f d \mu
        = \int_X f d \nu
        = \sum_{i \leq k} \alpha_i f(x_i).
    \]
\end{theorem}


\begin{corollary}\label{cor:richter-tchakaloff-frame}
    Let $X$ be a $G$-space where $X$ and $G$ are equipped with some Borel measures $\nu$ and $\mu$ respectively, and let $V \sub L^1(X)$ be a finite dimensional subspace with $\dim V = k$.
    Then for each $x \in X$, there exists a $k$-atomic measure $\upsilon_x = \sum_{i \leq k} \alpha_i \delta_{x_i}$ on $G$ such that for every $f \in V$,
    \[
        \int_G f (\inv{g} x) d \mu
        = \int_G f (\inv {g} x) d \upsilon_x
        = \sum_{i \leq k} \alpha_i f(\inv{g} x_i).
    \]
    \begin{proof}
        Since $V$ is finite dimensional, we can find $\set{e_i}_{i \leq k} \sub L^1(X)$ with $\Span \set{e_i}_{i \leq k} = V$.
        We can then define $\Hat{e}_i \in L^1(G)$ for $i \leq k$ by $\Hat{e}_i (g) := e(\inv{g} x)$.
        Now let $f \in V$, writing $f = \sum_{i \leq k} \alpha_i e_i$, and define $\Hat{f}(g) = f(\inv{g} x)$.
        So then setting $W := \Span \set{\Hat{e}_i} \sub L^1 (G)$, we have that $\dim W \leq \dim V$.
        We then have
        \[
            \Hat{f}(g)
            = f(\inv{g} x)
            = \sum_{i \leq k} \alpha_i e_i(\inv{g} x)
            = \sum_{i \leq k} \alpha_i \Hat{e}_i(g),
        \]
        so $\Hat{f} \in W$ for any $f \in V$.
        Hence, by theorem \ref{thm:richter-tchakaloff} there exists some $k$-atomic measure $\upsilon_x = \sum_i \alpha_i \delta_{g_i}$ on $G$ such that
        \[
            \int_G f(\inv{g} x) d \mu
            = \int_G \Hat{f}(g) d \mu
            = \int_G \Hat{f}(g) d \upsilon_x
            = \sum_i \alpha_i \Hat{f}(g_i)
            = \sum_i \alpha_i f(\inv{g_i} x).
        \]
    \end{proof}
\end{corollary}
