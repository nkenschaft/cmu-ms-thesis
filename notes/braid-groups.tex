\subsection{Braid Groups}\label{sec:bg-braid-groups}

\begin{definition}
    For a topological space $X$ and positive integer $n$,
    \[
        \Conf_n(X) := \set{(x_1,...,x_n) \in X^n : x_i \neq x_j \text{ for } i \neq j}
    \]
    is the space of $n$ ordered mutually distinct points in $X$.
\end{definition}

\begin{definition}
    The \emph{symmetric group on $n$ elements}, denoted $\Sigma_n$, is the set of all bijections on the first $n$ positive integers with the product given by function composition.
\end{definition}

\begin{definition}
    For a topological space $X$ and positive integer $n$,
    \[
        \UConf_n(X) := \Conf_n(X) / \Sigma_n = \set{\set{x_1,...,x_n} \sub X : x_i \neq x_j \text{ for } i \neq j}
    \]
    is the space of $n$ unordered mutually distinct points in $X$.
\end{definition}

\begin{definition}
    For a path-connected topological space $X$, the \emph{$n$-strand braid group on $X$} is
    \[
        B_n(X) := \pi_1(\UConf_n(X)).
    \]
    Similarly, the \emph{$n$-strand pure braid group on $X$} is
    \[
        P_n(X) := \pi_1(\Conf_n(X)).
    \]
    Note that these are equivalent to the classical definitions (see section 1.4 of \cite{kassel2008braid}).
\end{definition}

\begin{remark}
    The quotient map $q : \Conf_n(X) \to \UConf_n(X)$ is a covering space since $\Sigma_n$ acts freely and properly discontinuously on $\Conf_n(X)$.
\end{remark}

\begin{lemma}
    Let $q : \Conf_n(X) \to \UConf_n(X)$ be the quotient map.
    Then the induced map on fundamental groups $q_* : \pi_1(\Conf_n(X)) \to \pi_1(\UConf_n(X))$ is injective.
    \begin{proof}
        Let $\alpha, \beta : I \to \Conf_n(X)$ be paths such that $q_*([\alpha]) = q_*([\beta])$, i.e. $q \circ \alpha \simeq q \circ \beta$.
    \end{proof}
\end{lemma}
