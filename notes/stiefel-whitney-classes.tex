\subsection{Stiefel-Whitney Classes}\label{sec:bg-stiefel-whitney-classes}
The following material is from \cite{hatcher2009bundlesktheory}

\begin{definition}
    Let $E$ and $B$ be spaces and let $p : E \to B$ be a map such that
    \begin{enumerate}
        \item $p$ is surjective
        \item for all $b \in B$, $\inv{p}[\set{b}]$ has an $\R$-vector space structure,
        \item and there is a cover of $B$ by open sets $U_\alpha$ with homeomorphisms $\Phi_\alpha : \inv{p}[U_\alpha] \to U_\alpha \times \R^n$ such that $\Phi_\alpha |_{\inv{p}[\set{b}]} : \inv{p}[\set{b}] \to \set{b} \times \R^n$ is a linear isomorphism for each $b \in U_\alpha$.
    \end{enumerate}
    Then $p$ is an \emph{$n$-dimensional vector bundle} and the maps $\Phi_\alpha$ are \emph{local trivializations}.
\end{definition}

\begin{remark}
    An $n$-dimensional vector bundle is a special type of fiber bundle with fiber $\R^n$.
\end{remark}

\begin{proposition}\label{prop:trivial-bundle-iff-independent-sections}
    A vector bundle $p : E \to B$ is trivial (i.e. isomorphic to the product $B \times \R^n$) if and only there are $n$ linearly independent sections $s_1,...,s_k : B \to E$ of $p$.
\end{proposition}

\begin{definition}\label{def:pullback-bundle}
    Let $p : E \to B$ be a vector bundle and let $f : B' \to B$ be some map.
    Then we can define the \emph{pullback bundle} induced by $f$ to be
    \[
        E' = f^*(E) := \set{(b',e) \in B \times E : f(b') = p(e)}.
    \] 
\end{definition}

\begin{remark}
    From definition \ref{def:pullback-bundle}, the map $p' : E' \to B'$ given by $p'(b',e) = b'$ is also a vector bundle.
    Additionally, $E'$ is the unique pullback $B' \times_{B} E$ in the category $\Top$.
\end{remark}

\begin{definition}
    The \emph{Grassmann manifold} $\Gr(n,k)$ is the space of all $k$-dimensional vector subspaces of $\R^n$.
\end{definition}

\begin{remark}
    There is a natural vector bundle over $\Gr(n,k)$ given by
    \[
        E(n,k) := \set{(P,x) \in \Gr(n,k) \times \R^n : x \in P}
    \]
    with the projection $p : E(n,k) \to \Gr(n,k)$.
\end{remark}

\begin{definition}
    The \emph{Stiefel manifold} $\St(n,k)$ is the space of all orthonormal bases for $k$-dimensional vector subspaces of $\R^n$.
\end{definition}

\begin{remark}
    There is a natural vector bundle over $\St(n,k)$ given by
    \[
        F(n,k) := \set{(B,x) \in \St(n,k) \times \R^n : x \in \Span B}
    \]
    with the projection $q : F(n,k) \to \Gr(n,k)$.
    This bundle is trivial by proposition \ref{prop:trivial-bundle-iff-independent-sections} since one clearly has $k$ orthogonal (and hence linearly independent) sections of $q$ given by $s_i : \St(n,k) \to F(n,k)$, $s_i(B) = (B,b_i)$ where we write $B$ in the standard basis of $\R^n$ with $B = [b_1 \cdots b_k]$.
\end{remark}

\begin{remark}
    There is a natural projection $\pi : \St(n,k) \to \Gr(n,k)$ given by $\pi(B) = \Span B$.
\end{remark}

\begin{theorem}\label{thm:SW-class-existence}
    (Theorem 3.1 in \cite{hatcher2009bundlesktheory})
    There exist unique functions, the \emph{Stiefel-Whitney classes}, $w_1,w_2,...$ that map each real vector bundle $E \to B$ to a class $w_i(E) \in H^i(B; \Z_2)$ such that
    \begin{enumerate}
        \item If $E \cong E'$ as vector bundles, $w_i(E) = w_i(E')$.
        \item For a function $f : A \to B$ with induced map on homology $f^* : H^*(B;\Z_2) \to H^*(A;\Z_2)$ and pullback $f^*(E)$, we have $w_i(f^*(E)) = f^*(w_i(E))$.
        \item $w(E_1 \oplus E_2) = w(E_1) \smile w(E_2)$ where $w = 1 + w_1 + w_2 + \cdots \in H^*(B;\Z_2)$.
        \item $w_i(E) = 0$ if $i > \dim E$.
        \item For the canonical line bundle $E \to \R P^\infty$, $w_1(E)$ is a generator of $H^1(\R P^\infty;\Z_2)$.
    \end{enumerate}
\end{theorem}

\begin{remark}
    A trivial vector bundle has $w_i(E) = 0$ for $i > 0$.
\end{remark}

\begin{proposition}\label{prop:Grassmann-Z2-cohomology}
    (From \cite{jaworowski2006additive})
    Let $w_1,...,w_k$ be the Stiefel-Whitney classes of the bundle $p : E(n,k) \to \Gr(n,k)$, let $\Bar{w_1},...,\Bar{w_n}$ be the dual classes, and let $J(n,k)$ be the ideal in $\Z_2[w_1,...,w_k]$ generated by the relation $w \Bar{w} = 1$.
    Then
    \[
        H^*(\Gr(n,k)) \cong \Z_2[w_1,...,w_k]/J(n,k).
    \]
\end{proposition}

\begin{proposition}
    Let $\pi^*(E(n,k))$ be the pullback of the vector bundle $p : E(n,k) \to \Gr(n,k)$ via $\pi : \St(n,k) \to \Gr(n,k)$.
    Then the induced map on cohomology $\pi^* : H^*(\Gr(n,k)) \to H^*(\St(n,k))$ is the zero map, i.e. $\ker \pi^* = H^*(\Gr(n,k);\Z_2)$.
    \begin{proof}
        First observe that
        \begin{align*}
            \pi^*(E(n,k))
            &= \set{(B,(P,x)) \in \St(n,k) \times E(n,k) : \pi(B) = p(P,x) = P} \\
            &= \set{(B,x) \in \St(n,k) \times \R^n : x \in \Span B} \\
            &= F(n,k).
        \end{align*}
        Thus the pullback bundle is actually just the bundle $q : F(n,k) \to \St(n,k)$.
        Then by theorem \ref{thm:SW-class-existence} we have that $\pi^*(w_i(E(n,k))) = w_i(F(n,k)) = 0$ since $q : F(n,k) \to \St(n,k)$ is a trivial bundle.
        By proposition \ref{prop:Grassmann-Z2-cohomology} we know that $H^*(\Gr(n,k);\Z_2)$ is generated by (a quotient of) $w_1(E(n,k)),w_2(E(n,k)),...$, so we conclude that $\pi^*[H^*(\Gr(n,k);\Z_2)] = 0$.
    \end{proof}
\end{proposition}

\begin{corollary}
    $\nil \ker \pi^* = \cuplength(\Gr(n,k))$
\end{corollary}
