\subsection{Canonicalizing $\R^{d \times n}$ with respect to $\Sigma_n$}\label{sec:canonicalizing-matrices-permutations}
\begin{definition}
    The \emph{symmetric group on $n$ elements}, denoted $\Sigma_n$, is the set of all bijections on the first $n$ positive integers with the product given by function composition.
    For our purposes, we will equivalently consider $\Sigma_n$ as the set of $n$ by $n$ matrices with exactly one $1$ in each row and column and all other entries being $0$, with the product given by matrix multiplication.
\end{definition}
