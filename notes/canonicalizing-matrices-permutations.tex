\subsection{Canonicalizing $\R^{d \times n}$ with respect to $\Sigma_n$}\label{sec:canonicalizing-matrices-permutations}
With an extensive exploration of canonicalizing $\R^{d \times n}$ with respect to rotation, we now follow the lead of \cite{dym2024equivariant} and consider canonicalizing $\R^{d \times n}$ with respect to permutations.
More precisely, we wish to canonicalize $\R^{d \times n}$ with respect to $\Sigma_n$, the symmetric group on $n$ elements.
Practically speaking, this would allow us to operate on unordered collections of points in $\R^d$.

\begin{definition}
    The \emph{symmetric group on $n$ elements}, denoted $\Sigma_n$, is the set of all bijections on the first $n$ positive integers with the product given by function composition.
\end{definition}

\begin{definition}
    For a topological space $X$ and positive integer $n$,
    \[
        \Conf_n(X) := \set{(x_1,...,x_n) \in X^n : x_i \neq x_j \text{ for } i \neq j}
    \]
    is the space of $n$ ordered mutually distinct points in $X$.
\end{definition}

\begin{definition}
    For a topological space $X$ and positive integer $n$,
    \[
        \UConf_n(X) := \Conf_n(X) / \Sigma_n = \set{\set{x_1,...,x_n} \sub X : x_i \neq x_j \text{ for } i \neq j}
    \]
    is the space of $n$ unordered mutually distinct points in $X$.
\end{definition}

\begin{definition}
    A $G$-action on $X$ is \emph{properly discontinuous} if for every $x \in X$ there is an open neighborhood $U \ni x$ such that $g U \cap U = \emp$ for all $g \in G$.
\end{definition}

\begin{lemma}\label{lem:Hausdorff-free-prop-discont}
    Let $G$ be a finite group that acts freely on a Hausdorff space $X$.
    Then the action of $G$ is properly discontinuous.
    \begin{proof}
        Let $x \in X$ and write $G = \set{g_1,...,g_n}$ where $g_1 = e_G$.
        Since $X$ is Hausdorff, we can find open neighborhoods $U_1,...,U_n$ of $g_1 x,...,g_n x$ such that $U_i \cap U_j = \emp$ for $i \neq j$.
        \footnote{Got construction of $U$ \href{https://math.stackexchange.com/questions/3172278/finite-group-action-on-hausdorff-space}{here}}
        Now let $U = \cap_{i=2}^n \inv{g_i} U_i$ and observe that $U$ is an open set containing $x$ with $g_i U \sub U_i$.
        To finish, let $g \in G$ such that $gU \cap U \neq \emp$.
        Since $g_i U \cap U = \emp$ for $i \neq 1$ by construction, we conclude that $g = g_1$ and $G$ acts properly discontinuously on $X$.
    \end{proof}
\end{lemma}

\begin{lemma}\label{lem:permutation-quotient-covering}
    If $X$ is Hausdorff and $\Conf_n(X)$ is path-connected, then the quotient map $q : \Conf_n(X) \to \UConf_n(X)$ is a covering space.
    Additionally, $q$ is an $n!$-sheeted covering.
    \begin{proof}
        By proposition 1.40 of \cite{hatcher2005algebraic}, it suffices to show that the action of $\Sigma_n$ on $\Conf_n(X)$ is properly discontinuous, so by lemma \ref{lem:Hausdorff-free-prop-discont} it suffices to show that $\Sigma_n$ acts freely on $\Conf_n(X)$.
        However, $\Sigma_n$ clearly acts freely on $\Conf_n(X)$ by the definition of $\Conf_n(X)$, so $q$ is a covering map.

        For the second statement, proposition 1.40 of \cite{hatcher2005algebraic} also gives us that
        \[
            \Sigma_n \cong \pi_1(\UConf_n(X))/q_*(\pi_1(\Conf_n(X))),
        \]
        so by proposition 1.32 of \cite{hatcher2005algebraic} we conclude that the covering has $\card(\Sigma_n) = n!$ sheets.
    \end{proof}
\end{lemma}

\begin{remark}
    Observe that $\Conf_n(\R^d) \sub (\R^d)^n \cong \R^{d \times n}$ and consequently $\UConf_n(\R^d) \sub (\R^d)^n / \Sigma_n \cong \R^{d \times n} / \Sigma_n$.
\end{remark}

\begin{lemma}\label{lem:conf-Rd-pconn}
    For $d \geq 2$, $\Conf_n(\R^d)$ is path-connected.
    \begin{proof}
        Let $x,y \in \Conf_n(\R^d)$ and write $x = (x_1,...,x_n), y = (y_1,...,y_n)$.
        Then for every $i$, we can find some path
        \[
            \alpha_i : I \to \R^{d} \sminus \set{y_1,...,y_{i-1}, x_{i+1},...,x_n}
        \]
        such that $\alpha_i(0) = x_i$ and $\alpha_i(1) = y_i$ since $\R^d \sminus A$ for any finite set $A$ is still path-connected.
        Then we simply construct a path $\gamma : I \to \Conf_n(\R^d)$ with $\gamma(0) = x$ and $\gamma(1) = y$ by moving coordinate-wise via the $\alpha_i$'s, i.e.
        \[
            \gamma_i(t)
            = \begin{cases}
                x_i &\tif 0 \leq t < \f{i}{n} \\
                \alpha_i(nt - i) &\tif \f{i}{n} \leq t < \f{i+1}{n} \\
                y_i & \tif \f{i+1}{n} \leq t \leq 1.
            \end{cases}
        \]
        We can see that $\gamma(t) \in \Conf_n(\R^d)$ for all $t \in [0,1]$ since the $\alpha_i$'s are chosen such that when $\f{i}{n} \leq t < \f{i+1}{n}$, we have $\gamma_j(t) = y_j$ for $j < i$, $\gamma_k(t) = x_k$ for $k > i$, and $\gamma_i(t) = \alpha_i(nt-i) \not\in \set{y_1,...,y_{i-1},x_{i+1},...,x_n}$.
        Therefore we conclude that $\Conf_n(\R^d)$ is path-connected.
    \end{proof}
\end{lemma}

\begin{remark}
    Lemma \ref{lem:conf-Rd-pconn} fails for $\R$ since removing any points from $\R$ causes it to no longer be path-connected.
    In greater generality, lemma \ref{lem:conf-Rd-pconn} holds for any manifold $X$ with $\dim X \geq 2$, and fails if $\dim X = 1$ for the same reasons as for $\R$.
\end{remark}

\begin{lemma}\label{lem:cover-no-section}
    Let $p : \Tilde{X} \to X$ be a covering space where $\Tilde{X}$ is not homeomorphic to $X$ (i.e. the covering space has more than one sheet).
    Then there is no global section of $p$.
    \begin{proof}
        Assume for the sake of contradiction that we have a section $s : X \to \Tilde{X}$ of $p$, and fix some base point $x \in X$.
        Since $p$ has more than one sheet, we can find $x_1 \in \inv{p}(x)$ that is distinct from the base point $x_0 \in \Tilde{X}$.
        Then by path-connectivity we can find a path $\gamma : I \to X$ such that $\gamma(0) = x_0$ and $\gamma(1) = x_1$.
        Since $p(x_0) = p(x_1)$ by construction, we have that $p \circ \gamma$ is a loop in $X$ and $[p \circ \gamma]$ is a well-defined element of $\pi_1(X)$.
        Now observe that $p$ and $s$ induce maps $p_* : \pi_1(\Tilde{X}) \to \pi_1(X)$ and $s_* : \pi_1(X) \to \pi_1(\Tilde{X})$ such that $p_* \circ s_* = \id$.
        This means we can find $[\alpha] := s_*([p \circ \gamma]) \in \pi_1(\Tilde{X})$ such that $p_*([\alpha]) = [p \circ \alpha] = [p \circ \gamma]$, giving us a homotopy $h_t : I \to X$ such that $h_0 = p \circ \alpha$ and $h_1 = p \circ \gamma$.
        Since $\gamma$ clearly lifts $p \circ \gamma$, the homotopy lifting property gives us a unique homotopy lift $\Tilde{h_t} : I \to \Tilde{X}$ such that $p \circ \Tilde{h_t} = h_t$ with $\Tilde{h_1} = \gamma$.
        By the uniqueness of lifting we also have that $\alpha = \Tilde{h_0}$, but this is impossible since $\alpha$ is a loop and $\gamma$ is not.
        Therefore we conclude that there can be no global section $s$ of $p$.
    \end{proof}
\end{lemma}

\begin{remark}
    For completeness, we note that for $d = 1$, sorting is a canonicalization, and for $n = 1$ the identity is a canonicalization.
\end{remark}

As per with rotations, we would like to give some lower bounds for the sectional category of the relevant quotient map here.

\begin{theorem}\label{thm:permutation-secat-bounds}
    (Theorem 1.4 of \cite{roth2008configurationfibrations})
    Let $q : \Conf_n(\R^d) \to \Conf_n(\R^d) / \Sigma_n$ be the quotient map, and let $\alpha(n)$ be the number of $1$'s in the dyadic expansion of $n$.
    Then we have 
    \[
        (n - \alpha(n))(d-1) + 1
        \leq \secat(q)
        \leq \cat(\Conf_n(\R^d)/\Sigma_n)
        \leq (n-1)(d-1) + 1.
    \]
\end{theorem}

\begin{corollary}
    Let $q : \Conf_n(\R^d) \to \Conf_n(\R^d) / \Sigma_n$ be the quotient map.
    Then
    \[
        \secat(q) \geq (n - \ceil{\log_2(n)})(d-1) + 1
    \]
    where $\ceil{x}$ is the least integer greater than or equal to $x$.
\end{corollary}

By theorem \ref{thm:no-Sigman-canon-for-dxn} we cannot find a continuous canonicalization of $\R^{d \times n}$ with respect to $\Sigma_n$, but theorem \ref{thm:permutation-secat-bounds} tells us that, if we are willing to restrict to $\Conf_n(\R^d)$ we must be able to find a canonicalization with at most $(n-1)(d-1)+1$ open sets on which the canonicalization is continuous.
Unfortunately, the proof of this fact does not appear to be constructive, so the question remains how to produce such a canonicalization that could be used in practice.
